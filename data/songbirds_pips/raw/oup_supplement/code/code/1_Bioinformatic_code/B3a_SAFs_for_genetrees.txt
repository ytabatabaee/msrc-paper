## Bash code in this script represents bioinformatic processing tasks carried out on each species with possible shared PIP independently
## User needs to review output of PCA and lostruct scripts to identify PIPs that are mapped to the same chromosomes and might be shared
## User needs to create a set of metadata, here called "shared_pip_meta", including including its name, chromosome (reg), the start and stop loci of the PIP breakpoints, 
## and the start and stop loci of the COMP region the user has chosen
## Please note that the PIP and COMP breakpoints used here should apply as well as possible across all possibly shared PIPs across different species, and therefore
## are NOT necessarily the same breakpoints used in the "A" prefix scripts.
## The first part of the script uses an R script to pick which individuals will be included in gene trees.
## The bash variable $INVMETA represents a file the user must provide with columns indicating the species and "regn" (i.e. PIP name as used in the A prefix scripts) for each possible shared PIP,
## as well as a unique ID that applies to that shared PIP in a column called "shared.pip"
## The bash variable $GENOS represents the output of script B1.
## The bash variable $ALLSAMPS represents a text file the user must provide containing a full list of samples (with a column for species) available to be picked for gene trees. 
## To facilitate steps in script B3b, each individual that will be used in the gene tree needs a SAF file generated by ANGSD
## The bash variable $REF is a path to the corresponding reference genome.
## The user should prepare a list of bam files from all species used as as input for each possible shared PIP, represented by the bash variable "$BAMS"
## IMPORTANT: one component of this script involves random sampling of individuals, and it will therefore produce different results each time it is run
## This script is written as if it is in a SLURM array job where the variable $PIP is assigned to be different for each job in the array
## Proceed to script B3b to generate genetic distance estimates from the SAF files created here.

# ANGSD version: 0.941-6-g67b6b3b (htslib: 1.15-1-gf307dfb)
# winsfs version: 0.7.0
# R version: 4.2.0 

SPIP=$(awk -v pip=$PIP '$1==pip {print $1}' shared_pips_meta)
REG=$(awk -v pip=$PIP '$1==pip {print $2}' shared_pips_meta)
PSTART=$(awk -v pip=$PIP '$1==pip {print $3}' shared_pips_meta)
PSTOP=$(awk -v pip=$PIP '$1==pip {print $4}' shared_pips_meta)
CSTART=$(awk -v pip=$PIP '$1==pip {print $6}' shared_pips_meta)
CSTOP=$(awk -v pip=$PIP '$1==pip {print $7}' shared_pips_meta)



POUT=$PIP"_pip"


OUTCSV=$PIP"_selected_individuals.csv"

Rscript draw_pip_inds.R $BAMS $SPIP $PSTART $PSTOP $OUTCSV $REG $INVMETA $GENOS $ALLSAMPS



PSITES=$PIP"_pip.BED"

echo -e "$REG\t$PSTART\t$PSTOP" > $PSITES


angsd sites index $PSITES


CSITES=$REG"_comp.BED"


echo -e "$REG\t$CSTART\t$CSTOP" > $CSITES


angsd sites index $CSITES



while read BAM
do

SAMP=$(basename $BAM)


POUT=$PIP"_pip_"$SAMP

angsd -i $BAM -out $POUT -doSaf 1 -GL 2 \
-minMapQ 30 -minQ 30 -anc $REF \
-r $REG -sites $PSITES



if [[ $CSTART =~ ^-?[0-9]+$ ]] ;
then
echo "doing comp"


COUT=$PIP"_comp_"$SAMP



CSITES=$REG"_comp.BED"



angsd -i $BAM -out $COUT -doSaf 1 -GL 2 \
-minMapQ 30 -minQ 30 -anc $REF \
-r $REG -sites $CSITES



else
echo "not doing comp"
fi


done < $BAMS
