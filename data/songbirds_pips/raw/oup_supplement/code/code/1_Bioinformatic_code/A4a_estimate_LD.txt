## "species", "regn" (i.e. pip name), "sample" and "genotype"

## Please note that bioinformatic program commands are given according to how they work on the authors' computing cluster accounts. 
## Users replicating these steps would need to proceed according to how the programs are installed in their specific computing environment
## This code can be run on many species and many PIPs in parallel if it is formatted for a SLURM array job, for example.
## IMPORTANT: one component of this script involves random sampling, and it will therefore produce different results each time it is run
## Proceed to script A4b to create plots that are the basis of D4 and D6 in Fig. S1.

# I developed this function with interaction with chatGPT, which wrote the little comments within the function
balance_factor_sampling <- function(df, factor_column) {
  # Ensure that the specified column is a factor
  df[[factor_column]] <- as.factor(df[[factor_column]])
  
  # Calculate how much each factor level can contribute
  factor_levels <- levels(df[[factor_column]])
  n_levels <- length(factor_levels)
  counts <- table(df[[factor_column]])
  
  # Get the maximum counts across factor levels. This is also our target
  max_count <- max(counts)
  target_n <- max_count
  
  # Calculate the initial number of samples to take from each factor level
  sample_sizes <- rep(0, n_levels)
  names(sample_sizes) <- factor_levels
  min_count <- min(min(counts),  target_n %/% n_levels)
  for (lvl in factor_levels) {
    # Each level contributes proportionally, without exceeding max_count
    sample_sizes[lvl] <- min(counts[lvl], max(min_count, target_n %/% n_levels))
  }
  
  # Adjust if the sum of initial sample_sizes is less than target_n
  extra_needed <- target_n - sum(sample_sizes)
  while (extra_needed > 0) {
    for (lvl in factor_levels) {
      if (extra_needed == 0) break
      # We only add an extra sample if we have not exceeded the individual level count
      if (sample_sizes[lvl] < counts[lvl]) {
        sample_sizes[lvl] <- sample_sizes[lvl] + 1
        extra_needed <- extra_needed - 1
      }
    }
  }

resample <- function(x, ...) x[sample.int(length(x), ...)]
  
  # Subset the dataframe for each level and combine
  subsample_indices <- unlist(lapply(factor_levels, function(lvl) {
    if (sample_sizes[lvl] > 0) {
      resample(which(df[[factor_column]] == lvl), size = sample_sizes[lvl])
    }
  }))
  
  subsampled_df <- df[subsample_indices, , drop = FALSE]
  
  return(subsampled_df)
}
